TenantScopeMiddleware.php 
namespace App\Http\Middleware;

class InitializeTenantByDomainOrHeader
{
    public function handle($request, $next)
    {
        // Resolve tenant from custom header or subdomain
        $tenantId = $request->header('X-Tenant-ID') 
            ?? $this->resolveFromHost($request);

        if ($tenant = Tenant::find($tenantId)) {
            tenancy()->initialize($tenant);
            Context::add('tenant', $tenant->id);
        }

        return $next($request);
    }
}
